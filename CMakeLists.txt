cmake_minimum_required(VERSION 3.28)
project(sunjoo-auto-chroma VERSION 1.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(AC_OBS_MAJOR "33" CACHE STRING "OBS target: 32 or 33")
set_property(CACHE AC_OBS_MAJOR PROPERTY STRINGS 32 33)
if(NOT AC_OBS_MAJOR MATCHES "^(32|33)$")
  message(FATAL_ERROR "AC_OBS_MAJOR must be 32 or 33")
endif()

option(AC_BUILD_PLUGIN "Build the OBS filter and Qt dock" ON)
option(AC_BUILD_TESTS "Build algorithm regression tests" ON)
set(AC_SDK "${CMAKE_CURRENT_SOURCE_DIR}/.deps/obs-sdk-${AC_OBS_MAJOR}" CACHE PATH "OBS SDK root")
set(AC_OBS_INCLUDE_DIR "${AC_SDK}/include" CACHE PATH "libobs headers")
set(AC_OBS_LIBRARY "${AC_SDK}/lib/obs.lib" CACHE FILEPATH "OBS import library")
set(AC_FRONTEND_INCLUDE_DIR "${AC_SDK}/include" CACHE PATH "OBS frontend headers")
set(AC_FRONTEND_LIBRARY "${AC_SDK}/lib/obs-frontend-api.lib" CACHE FILEPATH "Frontend import library")
set(AC_LUA_LIBRARY "C:/Program Files/obs-studio/bin/64bit/lua51.dll" CACHE FILEPATH "LuaJIT runtime for tests")
list(PREPEND CMAKE_PREFIX_PATH "${CMAKE_CURRENT_SOURCE_DIR}/.deps/qt${AC_OBS_MAJOR}")
find_package(Qt6 REQUIRED COMPONENTS Core Widgets Concurrent)
set(AC_DATA "${CMAKE_CURRENT_BINARY_DIR}/data")
set(AC_VENDOR "${CMAKE_CURRENT_SOURCE_DIR}/native/vendor/auto-chroma-v0.3")
file(MAKE_DIRECTORY "${AC_DATA}/lua" "${AC_DATA}/locale")
set(AC_LUA_MODULES ac_roi ac_tune ac_tune_math ac_tune_despill ac_screen ac_diag ac_person ac_plate_math ac_plate_warn ac_autoplate ac_autoplate_img ac_ui)
foreach(module IN LISTS AC_LUA_MODULES)
  configure_file("${AC_VENDOR}/${module}.lua" "${AC_DATA}/lua/${module}.lua" COPYONLY)
endforeach()
foreach(resource ac_keyer.effect ac_plate_keyer.effect ac_plate_clean.effect mask.effect precision.effect region-shape.effectinc analysis.lua)
  configure_file("native/data/${resource}" "${AC_DATA}/${resource}" COPYONLY)
endforeach()
configure_file("${AC_VENDOR}/GPL-2.0.txt" "${AC_DATA}/GPL-2.0.txt" COPYONLY)
configure_file("${AC_VENDOR}/license.md" "${AC_DATA}/upstream-license.md" COPYONLY)
file(WRITE "${AC_DATA}/locale/ko-KR.ini" "Plugin.Name=\"자동 크로마키\"\n")
file(WRITE "${AC_DATA}/locale/en-US.ini" "Plugin.Name=\"Auto Chroma\"\n")
add_library(ac-analysis STATIC native/src/analyzer.cpp)
target_include_directories(ac-analysis PUBLIC native/src)
target_link_libraries(ac-analysis PUBLIC Qt6::Core)
target_compile_definitions(ac-analysis PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
  target_compile_options(ac-analysis PRIVATE /utf-8 /W4 /EHsc)
endif()
add_library(ac-precision STATIC native/src/precision-model.cpp)
target_include_directories(ac-precision PUBLIC native/src)
target_link_libraries(ac-precision PUBLIC Qt6::Gui)
if(MSVC)
  target_compile_options(ac-precision PRIVATE /utf-8 /W4 /EHsc)
endif()
if(AC_BUILD_PLUGIN)
  if(NOT EXISTS "${AC_OBS_INCLUDE_DIR}/obs.h")
    message(FATAL_ERROR "OBS SDK missing. Run scripts/prepare-deps.ps1 first.")
  endif()
  file(STRINGS "${AC_OBS_INCLUDE_DIR}/obs-config.h" _obs_version REGEX "^#define LIBOBS_API_MAJOR_VER +[0-9]+")
  if(NOT _obs_version MATCHES " ${AC_OBS_MAJOR}$")
    message(FATAL_ERROR "OBS SDK major mismatch: ${_obs_version}, expected ${AC_OBS_MAJOR}")
  endif()
  add_library(sunjoo-auto-chroma MODULE native/src/plugin-main.cpp native/src/filter.cpp native/src/dock.cpp
    native/src/preview-widget.cpp native/src/preview-driver.cpp native/src/scene-rotation.cpp)
  target_include_directories(sunjoo-auto-chroma PRIVATE native/src "${AC_OBS_INCLUDE_DIR}" "${AC_FRONTEND_INCLUDE_DIR}")
  target_link_libraries(sunjoo-auto-chroma PRIVATE ac-analysis ac-precision Qt6::Widgets Qt6::Concurrent "${AC_OBS_LIBRARY}" "${AC_FRONTEND_LIBRARY}")
  target_compile_definitions(sunjoo-auto-chroma PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN AC_OBS_TARGET=${AC_OBS_MAJOR})
  if(MSVC)
    target_compile_options(sunjoo-auto-chroma PRIVATE /utf-8 /W4 /EHsc)
  endif()
  set_target_properties(sunjoo-auto-chroma PROPERTIES PREFIX "" OUTPUT_NAME "sunjoo-auto-chroma")
  if(AC_OBS_MAJOR STREQUAL "33")
    set(AC_PLUGIN_ROOT "plugins/sunjoo-auto-chroma")
    install(TARGETS sunjoo-auto-chroma RUNTIME DESTINATION "${AC_PLUGIN_ROOT}" LIBRARY DESTINATION "${AC_PLUGIN_ROOT}")
  else()
    set(AC_PLUGIN_ROOT "obs-plugins/64bit")
    install(TARGETS sunjoo-auto-chroma RUNTIME DESTINATION "${AC_PLUGIN_ROOT}" LIBRARY DESTINATION "${AC_PLUGIN_ROOT}")
  endif()
  if(AC_OBS_MAJOR STREQUAL "33")
    install(DIRECTORY "${AC_DATA}/" DESTINATION "${AC_PLUGIN_ROOT}/data")
  else()
    install(DIRECTORY "${AC_DATA}/" DESTINATION "data/obs-plugins/sunjoo-auto-chroma")
  endif()
  install(FILES README.md native/THIRD-PARTY.md DESTINATION ".")
  install(FILES LICENSE DESTINATION "." RENAME GPL-2.0.txt)
endif()
if(AC_BUILD_TESTS)
  enable_testing()
  add_executable(ac-analyzer-test native/tests/analyzer-test.cpp)
  target_link_libraries(ac-analyzer-test PRIVATE ac-analysis)
  if(MSVC)
    target_compile_options(ac-analyzer-test PRIVATE /utf-8 /W4 /EHsc)
  endif()
  add_test(NAME analysis-regression COMMAND ac-analyzer-test "${AC_DATA}/lua" "${AC_LUA_LIBRARY}")
  add_executable(ac-precision-model-test native/tests/precision-model-test.cpp)
  target_link_libraries(ac-precision-model-test PRIVATE ac-precision Qt6::Core)
  if(MSVC)
    target_compile_options(ac-precision-model-test PRIVATE /utf-8 /W4 /EHsc)
  endif()
  add_test(NAME precision-model-regression COMMAND ac-precision-model-test)
  if(AC_BUILD_PLUGIN)
    add_executable(ac-scene-rotation-test native/tests/scene-rotation-test.cpp native/src/scene-rotation.cpp)
    target_include_directories(ac-scene-rotation-test PRIVATE native/src "${AC_OBS_INCLUDE_DIR}" "${AC_FRONTEND_INCLUDE_DIR}")
    target_compile_definitions(ac-scene-rotation-test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(ac-scene-rotation-test PRIVATE Qt6::Gui)
    if(MSVC)
      target_compile_options(ac-scene-rotation-test PRIVATE /utf-8 /W4 /EHsc)
    endif()
    add_test(NAME scene-rotation-regression COMMAND ac-scene-rotation-test)
  endif()
  add_executable(ac-preview-test native/tests/preview-test.cpp native/src/preview-widget.cpp)
  target_include_directories(ac-preview-test PRIVATE native/src)
  target_link_libraries(ac-preview-test PRIVATE Qt6::Widgets)
  if(MSVC)
    target_compile_options(ac-preview-test PRIVATE /utf-8 /W4 /EHsc)
  endif()
  add_test(NAME preview-interaction-regression COMMAND ac-preview-test)
  get_target_property(_qtcore Qt6::Core IMPORTED_LOCATION_RELEASE)
  get_filename_component(_qtbin "${_qtcore}" DIRECTORY)
  get_filename_component(_luabin "${AC_LUA_LIBRARY}" DIRECTORY)
  set_tests_properties(analysis-regression PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${_qtbin};PATH=path_list_prepend:${_luabin}")
  set_tests_properties(precision-model-regression PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${_qtbin}")
  if(AC_BUILD_PLUGIN)
    set_tests_properties(scene-rotation-regression PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${_qtbin}")
  endif()
  set_tests_properties(preview-interaction-regression PROPERTIES TIMEOUT 20
    ENVIRONMENT "QT_PLUGIN_PATH=${_qtbin}/../plugins"
    ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${_qtbin}")
endif()

# Optional real-device integration tests. Requires an OBS runtime matching AC_OBS_MAJOR.
option(AC_BUILD_GPU_TESTS "Build real D3D11 filter integration tests" OFF)
set(AC_GPU_RUNTIME_BIN "C:/Program Files/obs-studio/bin/64bit" CACHE PATH "Matching OBS runtime DLL directory")
set(AC_GPU_LIBOBS_DATA "C:/Program Files/obs-studio/data/libobs" CACHE PATH "Matching libobs effect data")
if(AC_BUILD_GPU_TESTS AND AC_BUILD_PLUGIN)
  enable_testing()
  add_executable(ac-gpu-test native/tests/gpu-test.cpp native/src/filter.cpp)
  target_include_directories(ac-gpu-test PRIVATE native/src "${AC_OBS_INCLUDE_DIR}")
  target_link_libraries(ac-gpu-test PRIVATE ac-precision Qt6::Core Qt6::Gui "${AC_OBS_LIBRARY}")
  target_compile_definitions(ac-gpu-test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  if(MSVC)
    target_compile_options(ac-gpu-test PRIVATE /utf-8 /W4 /EHsc)
  endif()
  add_dependencies(ac-gpu-test sunjoo-auto-chroma)
  add_test(NAME gpu-filter-integration COMMAND ac-gpu-test "$<TARGET_FILE:sunjoo-auto-chroma>" "${AC_DATA}" "${AC_GPU_LIBOBS_DATA}" "$<TARGET_FILE_DIR:Qt6::Core>")
  get_target_property(_ac_gpu_qtcore Qt6::Core IMPORTED_LOCATION_RELEASE)
  get_filename_component(_ac_gpu_qtbin "${_ac_gpu_qtcore}" DIRECTORY)
  set_tests_properties(gpu-filter-integration PROPERTIES TIMEOUT 60
    ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${_ac_gpu_qtbin};PATH=path_list_prepend:${AC_GPU_RUNTIME_BIN}")
  add_executable(ac-sample-test native/tests/sample-test.cpp native/src/filter.cpp)
  target_include_directories(ac-sample-test PRIVATE native/src "${AC_OBS_INCLUDE_DIR}")
  target_link_libraries(ac-sample-test PRIVATE ac-precision Qt6::Core Qt6::Gui "${AC_OBS_LIBRARY}")
  target_compile_definitions(ac-sample-test PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  if(MSVC)
    target_compile_options(ac-sample-test PRIVATE /utf-8 /W4 /EHsc)
  endif()
  add_dependencies(ac-sample-test sunjoo-auto-chroma)
  set(AC_SAMPLE_IMAGE "${CMAKE_CURRENT_SOURCE_DIR}/.analysis/sample-review/raw-matched.png" CACHE FILEPATH "Extracted supplied sample frame")
  if(EXISTS "${AC_SAMPLE_IMAGE}")
    add_test(NAME supplied-sample-gpu COMMAND ac-sample-test "$<TARGET_FILE:sunjoo-auto-chroma>" "${AC_DATA}"
      "${AC_GPU_LIBOBS_DATA}" "$<TARGET_FILE_DIR:Qt6::Core>" "${AC_SAMPLE_IMAGE}" "${CMAKE_CURRENT_BINARY_DIR}/sample-evidence")
    set_tests_properties(supplied-sample-gpu PROPERTIES TIMEOUT 90
      ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${_ac_gpu_qtbin};PATH=path_list_prepend:${AC_GPU_RUNTIME_BIN}")
  endif()
endif()
